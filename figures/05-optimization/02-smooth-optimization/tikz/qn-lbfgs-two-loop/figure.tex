\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}

\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}

\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont},
pair/.style={draw=VB,fill=VBF,rounded corners=1.8pt,minimum width=20mm,minimum height=9.5mm,inner sep=1pt,align=center},
flow/.style={->,line width=.95pt},end/.style={draw=VL,rounded corners=1.8pt,minimum width=12mm,minimum height=8mm,inner sep=1pt}]
\node[pair] (a) at (30,0) {$\bm{s}^{(t-3)},\bm{y}^{(t-3)}$};
\node[pair] (b) at (54,0) {$\bm{s}^{(t-2)},\bm{y}^{(t-2)}$};
\node[pair] (c) at (78,0) {$\bm{s}^{(t-1)},\bm{y}^{(t-1)}$};
\node[end,draw=VG,fill=VGF] (m) at (8,0) {$\bm{z}=\gamma\bm{q}$};
\node[end] (g) at (101,12) {$\bm{g}^{(t)}$};
\node[end] (o) at (101,-12) {$\bm{W}^{(t)}\bm{g}^{(t)}$};
\coordinate (t1) at (78,12); \coordinate (t2) at (54,12); \coordinate (t3) at (30,12);

\draw[flow,VY] (g.west)--(8,12)--(m.north);
\draw[flow,VP] (m.south)--(8,-12)--(o.west);
\foreach \p in {t1,t2,t3}{\fill[VY] (\p) circle (.9);}
\foreach \p in {b1,b2,b3}{\fill[VP] (\p) circle (.9);}

\draw[VP,line width=.6pt] (b1)--(a.south); \draw[VP,line width=.6pt] (b2)--(b.south); \draw[VP,line width=.6pt] (b3)--(c.south);
\node[text=VI,anchor=south] at (54,13) {第一层：由新到旧，得到$\xi_j$，从$\bm{q}$中减去$\xi_j\bm{y}^{(j)}$};
\node[text=VI,anchor=north] at (54,-13) {第二层：由旧到新，得到$\eta_j$，给$\bm{z}$加上$(\xi_j-\eta_j)\bm{s}^{(j)}$};
\end{tikzpicture}
