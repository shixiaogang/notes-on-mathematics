% Dijkstra 出队轨迹：s->a 2, s->b 5, a->b 1, a->t 5, b->t 2（Python 验算）。
% 出队：s(0)：a=2,b=5；a(2)：b 5->3, t=7；b(3)：t 7->5；t(5)。
\begin{tikzpicture}[>=Stealth,
  v/.style={circle,draw=FigureStroke,fill=white,line width=.8pt,minimum size=6mm,inner sep=0pt},
  vdone/.style={v,fill=FigureBlueFill},
  vcur/.style={v,draw=FigureYellow,fill=FigureYellowFill},
  arc/.style={->,draw=FigureGrid,line width=.8pt},
  pred/.style={->,draw=FigureBlue,line width=1.4pt},
  newp/.style={->,draw=FigureGreen,line width=1.4pt},
  w/.style={fill=white,inner sep=.6pt,font=\figurefont\fontsize{7}{8}\selectfont},
  lab/.style={font=\figurefont\fontsize{7.5}{9}\selectfont},
  imp/.style={lab,text=FigureGreen}]
\newcommand{\pos}{%
  \coordinate (S) at (0,0); \coordinate (A) at (1.45,1.0);
  \coordinate (B) at (1.45,-1.0); \coordinate (T) at (2.9,0);}
% 面板 1：取出 s
\begin{scope}
  \pos
  \node[vcur] (s) at (S) {$s$}; \node[v] (a) at (A) {$a$}; \node[v] (b) at (B) {$b$}; \node[v] (t) at (T) {$t$};
  \draw[newp] (s)--node[w]{$2$}(a); \draw[newp] (s)--node[w]{$5$}(b);
  \draw[arc] (a)--node[w]{$1$}(b); \draw[arc] (a)--node[w]{$5$}(t); \draw[arc] (b)--node[w]{$2$}(t);
  \node[lab,above=.5mm of s] {$0$};
  \node[imp,above=.5mm of a] {$2$}; \node[imp,below=.5mm of b] {$5$}; \node[lab,above=.5mm of t] {$\infty$};
\end{scope}
\end{tikzpicture}
