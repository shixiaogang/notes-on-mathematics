% Same graph, two arrangements of two unit readings. Edge labels are w(du)^2.
\begin{tikzpicture}[x=1mm,y=1mm,
  font=\figurefont\fontsize{8}{10}\selectfont,text=FigureInk,
  vertex/.style={circle,draw=FigureStroke,line width=.65pt,fill=white,
    minimum size=6mm,inner sep=0pt},
  one/.style={vertex,fill=FigureBlueFill},
  active/.style={draw=FigureBlue,line width=1.05pt},
  inactive/.style={draw=FigureMuted,line width=.7pt,dashed},
  term/.style={fill=white,inner sep=1pt}]
  \def\leftpanel{a}
  \foreach \offset/\panel in {0/a,44/b}{
    \begin{scope}[xshift=\offset mm]
      \ifx\panel\leftpanel
        \node[one] (v3) at (22,0) {$1$};
        \node[vertex] (v2) at (22,20) {$0$};
      \else
        \node[vertex] (v3) at (22,0) {$0$};
        \node[one] (v2) at (22,20) {$1$};
      \fi
      \node[one] (v1) at (0,20) {$1$};
      \node[vertex] (v4) at (0,0) {$0$};
      \draw[active] (v2)--node[right,term] {$1$} (v3);
      \draw[active] (v4)--node[left,term] {$1$} (v1);
      \ifx\panel\leftpanel
        \draw[active] (v1)--node[above,term] {$1$} (v2);
        \draw[active] (v3)--node[below,term] {$1$} (v4);
        \draw[inactive] (v1)--node[inner sep=.8pt,above right] {$0$} (v3);
        \node at (11,-9) {$\bm x^\top\bm L\bm x=4$};
      \else
        \draw[inactive] (v1)--node[above,term] {$0$} (v2);
        \draw[inactive] (v3)--node[below,term] {$0$} (v4);
        \draw[active] (v1)--node[inner sep=.8pt,above right] {$1$} (v3);
        \node at (11,-9) {$\bm x^\top\bm L\bm x=3$};
      \fi
      \node at (11,-14.5) {(\panel)};
    \end{scope}
  }
\end{tikzpicture}
