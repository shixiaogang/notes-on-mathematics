% Three fundamental cuts obtained by deleting tree edges 12, 23, 34 from T={12,23,34}.
\begin{tikzpicture}[x=1mm,y=1mm,font=\figurefont\fontsize{8}{10}\selectfont,
  vertex/.style={circle,draw=FigureBlue,fill=FigureBlueFill,
      minimum size=4.6mm,inner sep=0pt,line width=.55pt},
  quiet/.style={-{Stealth[length=1.55mm]},draw=FigureMuted,dashed,line width=.58pt},
  cut/.style={-{Stealth[length=1.65mm]},draw=FigureGreen,line width=.78pt},
  cutrev/.style={-{Stealth[length=1.65mm]},draw=FigureRed,line width=.78pt},
  panel/.style={font=\figurefont\fontsize{7.5}{9}\selectfont,text=FigureInk},
  reading/.style={fill=white,text=FigureInk,inner sep=.7pt}]
  % Panel (a): delete 12 -> S_1 = {1}, d_1 = (1,0,0,-1,1)^T
  \begin{scope}[xshift=0mm]
    \fill[FigureBlueFill,rounded corners=1.8pt] (-4,12.5) rectangle (4,20.5);
    \fill[FigureGreenFill,rounded corners=1.8pt]
      (-4,-4) -- (22,-4) -- (22,20.5) -- (13.5,20.5) -- (13.5,4.5) -- (-4,4.5) -- cycle;
    \draw[FigureMuted,dashed,line width=.55pt] (-5.2,8.5) -- (8.8,8.5) -- (8.8,21.2);
    \node[vertex] (a1) at (0,16.5) {$1$};
    \node[vertex] (a2) at (18,16.5) {$2$};
    \node[vertex] (a3) at (18,0) {$3$};
    \node[vertex] (a4) at (0,0) {$4$};
    \draw[cut] (a1) -- node[reading,above] {$+1$} (a2);
    \draw[quiet] (a2) -- node[reading,right] {$0$} (a3);
    \draw[quiet] (a3) -- node[reading,below] {$0$} (a4);
    \draw[cutrev] (a4) -- node[reading,left] {$-1$} (a1);
    \draw[cut] (a1) -- node[reading,above right] {$+1$} (a3);
    \node[panel] at (9,-11) {(a) 删 $12$ 得 $\bm d_1$};
  \end{scope}

  % Panel (b): delete 23 -> S_2 = {1,2}, d_2 = (0,1,0,-1,1)^T
  \begin{scope}[xshift=36mm]
    \fill[FigureBlueFill,rounded corners=1.8pt] (-4,12.5) rectangle (22,20.5);
    \fill[FigureGreenFill,rounded corners=1.8pt] (-4,-4) rectangle (22,4);
    \draw[FigureMuted,dashed,line width=.55pt] (-5.2,8.25) -- (23.2,8.25);
    \node[vertex] (b1) at (0,16.5) {$1$};
    \node[vertex] (b2) at (18,16.5) {$2$};
    \node[vertex] (b3) at (18,0) {$3$};
    \node[vertex] (b4) at (0,0) {$4$};
    \draw[quiet] (b1) -- node[reading,above] {$0$} (b2);
    \draw[cut] (b2) -- node[reading,right] {$+1$} (b3);
    \draw[quiet] (b3) -- node[reading,below] {$0$} (b4);
    \draw[cutrev] (b4) -- node[reading,left] {$-1$} (b1);
    \draw[cut] (b1) -- node[reading,above right] {$+1$} (b3);
    \node[panel] at (9,-11) {(b) 删 $23$ 得 $\bm d_2$};
  \end{scope}

  % Panel (c): delete 34 -> S_3 = {1,2,3}, d_3 = (0,0,1,-1,0)^T
  \begin{scope}[xshift=72mm]
    \fill[FigureBlueFill,rounded corners=1.8pt]
      (-4,12) -- (-4,20.5) -- (22,20.5) -- (22,-4) -- (13.5,-4) -- (13.5,12) -- cycle;
    \fill[FigureGreenFill,rounded corners=1.8pt] (-4,-4) rectangle (4,4);
    \draw[FigureMuted,dashed,line width=.55pt] (-5.2,8) -- (8.8,8) -- (8.8,-4.8);
    \node[vertex] (c1) at (0,16.5) {$1$};
    \node[vertex] (c2) at (18,16.5) {$2$};
    \node[vertex] (c3) at (18,0) {$3$};
    \node[vertex] (c4) at (0,0) {$4$};
    \draw[quiet] (c1) -- node[reading,above] {$0$} (c2);
    \draw[quiet] (c2) -- node[reading,right] {$0$} (c3);
    \draw[cut] (c3) -- node[reading,below] {$+1$} (c4);
    \draw[cutrev] (c4) -- node[reading,left] {$-1$} (c1);
    \draw[quiet] (c1) -- node[reading,above right] {$0$} (c3);
    \node[panel] at (9,-11) {(c) 删 $34$ 得 $\bm d_3$};
  \end{scope}
\end{tikzpicture}
