% Top row: edges used. Each lower row is a distinct sequence of five visits.
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,
  vertex/.style={circle,draw=FigureBlue,fill=FigureBlueFill,inner sep=0pt,
    minimum size=4.3mm,line width=.52pt},
  visit/.style={vertex,minimum size=3.75mm},
  endpoint/.style={visit,draw=FigureYellow,fill=FigureYellowFill,line width=.72pt},
  edge/.style={draw=FigureStroke,line width=.65pt},
  walkstep/.style={edge,-{Stealth[length=1.1mm,width=.9mm]}},
  panel/.style={anchor=mid,font=\figurefont\footnotesize}]
  \newcommand{\WalkVisits}[2]{%
    \begin{scope}[yshift=#1mm]
      \foreach \letter [count=\slot from 0] in {#2} {
        \ifnum\slot=0
          \node[endpoint] (visit-\slot) at (6*\slot,0) {$\letter$};
        \else
          \ifnum\slot=4
            \node[endpoint] (visit-\slot) at (6*\slot,0) {$\letter$};
          \else
            \node[visit] (visit-\slot) at (6*\slot,0) {$\letter$};
          \fi
          \pgfmathtruncatemacro{\previous}{\slot-1}
          \draw[walkstep] (visit-\previous) -- (visit-\slot);
        \fi
      }
    \end{scope}
  }
  \begin{scope}
    \node[vertex] (a) at (5.5,0) {$u$};
    \node[vertex] (b) at (18.5,0) {$v$};
    \draw[edge] (a) -- (b);
    \WalkVisits{-12}{u,v,u,v,u}
    \WalkVisits{-18}{v,u,v,u,v}
    \node[panel] at (12,-35) {(a)};
  \end{scope}
  \begin{scope}[xshift=34mm]
    \node[vertex] (a) at (3,-2) {$a$};
    \node[vertex] (b) at (12,4) {$v$};
    \node[vertex] (c) at (21,-2) {$b$};
    \draw[edge] (a) -- (b) -- (c);
    \WalkVisits{-12}{v,a,v,b,v}
    \WalkVisits{-18}{v,b,v,a,v}
    \WalkVisits{-24}{a,v,b,v,a}
    \WalkVisits{-30}{b,v,a,v,b}
    \node[panel] at (12,-35) {(b)};
  \end{scope}
  \begin{scope}[xshift=68mm]
    \node[vertex] (u) at (7,5) {$u$};
    \node[vertex] (v) at (17,5) {$v$};
    \node[vertex] (w) at (17,-5) {$w$};
    \node[vertex] (z) at (7,-5) {$z$};
    \draw[edge] (u) -- (v) -- (w) -- (z) -- (u);
    \WalkVisits{-12}{u,v,w,z,u}
    \node[panel] at (12,-35) {(c)};
  \end{scope}
\end{tikzpicture}
