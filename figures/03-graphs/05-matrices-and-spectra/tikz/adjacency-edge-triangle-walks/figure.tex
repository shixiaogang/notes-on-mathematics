% Each arrow label gives its time index within the illustrated walk.
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,
  vertex/.style={circle,draw=FigureBlue,fill=FigureBlueFill,inner sep=0pt,
    minimum size=4.3mm,line width=.52pt},
  origin/.style={vertex,fill=FigureYellowFill,draw=FigureYellow,line width=.72pt},
  walkstep/.style={->,draw=FigureStroke,line width=.67pt},
  time/.style={font=\figurefont\fontsize{7.5}{9}\selectfont,inner sep=.8pt},
  panel/.style={anchor=mid,font=\figurefont\footnotesize}]
  \foreach \offset/\first/\last/\paneltag in {0/u/v/a,24/v/u/b} {
    \begin{scope}[xshift=\offset mm]
      \node[origin] (from) at (0,0) {$\first$};
      \node[vertex] (through) at (13,0) {$\last$};
      \draw[walkstep] (from) to[bend left=32] node[time,above] {$1$} (through);
      \draw[walkstep] (through) to[bend left=32] node[time,below] {$2$} (from);
      \node[panel] at (6.5,-12) {(\paneltag)};
    \end{scope}
  }
  \foreach \offset/\paneltag in {49/c,73/d} {
    \begin{scope}[xshift=\offset mm]
      \node[origin] (u) at (6.5,6) {$u$};
      \node[vertex] (l) at (0,-4) {$v$};
      \node[vertex] (r) at (13,-4) {$w$};
      \ifnum\offset=49
        \draw[walkstep] (u) -- node[time,above left] {$1$} (l);
        \draw[walkstep] (l) -- node[time,below] {$2$} (r);
        \draw[walkstep] (r) -- node[time,above right] {$3$} (u);
      \else
        \draw[walkstep] (u) -- node[time,above right] {$1$} (r);
        \draw[walkstep] (r) -- node[time,below] {$2$} (l);
        \draw[walkstep] (l) -- node[time,above left] {$3$} (u);
      \fi
      \node[panel] at (6.5,-12) {(\paneltag)};
    \end{scope}
  }
\end{tikzpicture}
