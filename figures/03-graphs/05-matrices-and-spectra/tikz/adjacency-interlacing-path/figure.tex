% The horizontal positions are proportional to the eigenvalues.
\begin{tikzpicture}[x=1mm,y=1mm,
  vertex/.style={circle,draw=FigureBlue,fill=FigureBlueFill,inner sep=0pt,
    minimum size=4.3mm,line width=.52pt},
  edge/.style={draw=FigureStroke,line width=.65pt},
  old/.style={circle,draw=FigureBlue,fill=FigureBlueFill,line width=.56pt,inner sep=0pt,minimum size=2.5mm},
  new/.style={rectangle,draw=FigureGreen,fill=FigureGreenFill,line width=.56pt,inner sep=0pt,minimum size=2.5mm}]
  \pgfmathsetmacro{\specbound}{sqrt(2)*21}
  \fill[FigureBlueFill] (-\specbound,-12) rectangle (0,3);
  \fill[FigureGreenFill] (0,-12) rectangle (\specbound,3);
  \draw[draw=FigureMuted,dashed,line width=.52pt]
    (-\specbound,-12) -- (-\specbound,3) (0,-12) -- (0,3) (\specbound,-12) -- (\specbound,3);
  \draw[edge] (-34,0) -- (34,0) (-34,-9) -- (34,-9);
  \foreach \position/\spectrumlabel in {-\specbound/{-\sqrt{2}},0/0,\specbound/{\sqrt{2}}} {
    \node[old] at (\position,0) {};
    \node[above=1.8mm] at (\position,0) {$\spectrumlabel$};
  }
  \foreach \position/\spectralvalue in {-21/-1,21/1} {
    \node[new] at (\position,-9) {};
    \node[below=1.8mm] at (\position,-9) {$\spectralvalue$};
  }

  \foreach \level in {0,-9} {
    \node[vertex] (one) at (-53,\level) {$1$};
    \node[vertex] (two) at (-46,\level) {$2$};
    \draw[edge] (one) -- (two);
    \ifnum\level=0
      \node[vertex] (three) at (-39,\level) {$3$};
      \draw[edge] (two) -- (three);
    \fi
  }
\end{tikzpicture}
