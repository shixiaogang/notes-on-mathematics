% Fractional and integer assignments on C4 and C3.
\begin{tikzpicture}[x=1mm,y=1mm,font=\figurefont\fontsize{8}{10}\selectfont,
  vertex/.style={circle,draw=FigureBlue,fill=FigureBlueFill,
    minimum size=4.6mm,inner sep=0pt,line width=.55pt},
  used/.style={draw=FigureBlue,line width=.86pt},
  unused/.style={draw=FigureMuted,dashed,line width=.55pt},
  value/.style={text=FigureInk,inner sep=1pt},
  panel/.style={font=\figurefont\fontsize{7.5}{9}\selectfont,text=FigureInk}]
  \foreach \shift/\paneltag in {0/{(a) 分数方案},27/{(b) 整数方案}} {
    \begin{scope}[xshift=\shift mm]
      \node[vertex] (a) at (0,14) {$1$};
      \node[vertex] (b) at (13,14) {$2$};
      \node[vertex] (c) at (13,0) {$3$};
      \node[vertex] (d) at (0,0) {$4$};
      \ifnum\shift=0
        \draw[used] (a) -- node[value,above] {$\tfrac12$} (b)
                          -- node[value,right] {$\tfrac12$} (c)
                          -- node[value,below] {$\tfrac12$} (d)
                          -- node[value,left] {$\tfrac12$} (a);
      \else
        \draw[used] (a) -- node[value,above] {$1$} (b);
        \draw[used] (c) -- node[value,below] {$1$} (d);
        \draw[unused] (b) -- node[value,right] {$0$} (c);
        \draw[unused] (d) -- node[value,left] {$0$} (a);
      \fi
      \node[panel] at (6.5,-11) {\paneltag};
    \end{scope}
  }
  \foreach \shift/\paneltag in {55/{(c) 分数方案},81/{(d) 整数方案}} {
    \begin{scope}[xshift=\shift mm]
      \node[vertex] (a) at (0,0) {$1$};
      \node[vertex] (b) at (6.5,14) {$2$};
      \node[vertex] (c) at (13,0) {$3$};
      \ifnum\shift=55
        \draw[used] (a) -- node[value,above left] {$\tfrac12$} (b)
                          -- node[value,above right] {$\tfrac12$} (c)
                          -- node[value,below] {$\tfrac12$} (a);
      \else
        \draw[used] (a) -- node[value,above left] {$1$} (b);
        \draw[unused] (b) -- node[value,above right] {$0$} (c);
        \draw[unused] (c) -- node[value,below] {$0$} (a);
      \fi
      \node[panel] at (6.5,-11) {\paneltag};
    \end{scope}
  }
\end{tikzpicture}
