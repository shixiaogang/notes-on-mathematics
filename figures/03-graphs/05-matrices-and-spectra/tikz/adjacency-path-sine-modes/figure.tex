% Sine eigenmodes on P_3 with virtual zero endpoints at j=0 and j=4.
\begin{tikzpicture}[x=1mm,y=1mm,font=\figurefont\fontsize{8}{10}\selectfont,
  vertex/.style={circle,draw=FigureBlue,fill=FigureBlueFill,
    minimum size=3.8mm,inner sep=0pt,line width=.55pt},
  virtual/.style={circle,draw=FigureMuted,fill=white,dashed,
    minimum size=3.8mm,inner sep=0pt,line width=.5pt,text=FigureMuted},
  stem/.style={draw=FigureBlue,line width=.7pt},
  curve/.style={draw=FigureBlue!70!white,line width=.65pt},
  dot/.style={circle,fill=FigureBlue,inner sep=0pt,minimum size=1.6mm},
  vdot/.style={circle,draw=FigureMuted,fill=white,inner sep=0pt,minimum size=1.5mm,line width=.5pt},
  axis/.style={draw=FigureStroke,line width=.55pt},
  note/.style={font=\figurefont\fontsize{7}{8.5}\selectfont,text=FigureMuted},
  panel/.style={font=\figurefont\fontsize{7.5}{9}\selectfont,text=FigureInk}]
  \foreach \shift/\kval/\ptitle in {
      0/1/{(a) $k=1,\ \lambda_1=\sqrt{2}$},
      35/2/{(b) $k=2,\ \lambda_2=0$},
      70/3/{(c) $k=3,\ \lambda_3=-\sqrt{2}$}} {
    \begin{scope}[xshift=\shift mm]
      % Baseline axis y=0
      \draw[FigureMuted,dashed,line width=.45pt] (0,0) -- (24,0);
      % Smooth sine curve
      \draw[curve,domain=0:4,samples=60,smooth]
        plot ({6*\x}, {8.5*sin(deg(\x*\kval*pi/4))});
      % Stems and mode values at j=1,2,3
      \foreach \j in {1,2,3} {
        \pgfmathsetmacro{\yval}{8.5*sin(deg(\j*\kval*pi/4))}
        \draw[stem] ({6*\j},0) -- ({6*\j},\yval);
        \node[dot] at ({6*\j},\yval) {};
      }
      \node[vdot] at (0,0) {};
      \node[vdot] at (24,0) {};
      % Path vertices drawn below at y=-14
      \draw[FigureMuted,dashed,line width=.45pt] (0,-14) -- (6,-14) (18,-14) -- (24,-14);
      \draw[axis] (6,-14) -- (18,-14);
      \node[virtual] at (0,-14) {$0$};
      \node[vertex] at (6,-14) {$1$};
      \node[vertex] at (12,-14) {$2$};
      \node[vertex] at (18,-14) {$3$};
      \node[virtual] at (24,-14) {$4$};
      \node[panel] at (12,-22) {\ptitle};
    \end{scope}
  }
\end{tikzpicture}
