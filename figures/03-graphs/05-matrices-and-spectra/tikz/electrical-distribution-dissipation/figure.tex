\begin{tikzpicture}[
  x=1mm,y=1mm,
  terminal/.style={circle,draw=FigureBlue,fill=FigureBlueFill,minimum size=4.5mm,inner sep=0pt,line width=.65pt},
  internal/.style={terminal,draw=FigureStroke,fill=white},
  conductor/.style={draw=FigureMuted,dashed,line width=.65pt},
  current/.style={-{Stealth[length=1.6mm,width=1.1mm]},draw=FigureBlue,line width=.9pt},
  amount/.style={font=\FigureNoteFont\fontsize{7.5}{9}\selectfont,fill=white,inner sep=.5pt},
  panelmark/.style={font=\figurefont\fontsize{8}{10}\selectfont,anchor=north,inner sep=0pt},
]
  \def\zeroflow{0}
  \begin{scope}[local bounding box=panelrow]
    \foreach \x/\case/\topflow/\midflow/\bottomflow/\energy in
        {0/a/0/1/0/1,34/b/{1/2}/{1/2}/0/{3/4},68/c/{1/4}/{1/2}/{1/4}/{1/2}}{
      \begin{scope}[xshift=\x mm,local bounding box=panel\case]
        \node[terminal] (l) at (-11,0) {$1$};
        \node[terminal] (r) at (11,0) {$3$};
        \node[internal] (u) at (0,9) {$2$};
        \node[internal] (d) at (0,-9) {$4$};
        \ifx\topflow\zeroflow
          \draw[conductor] (l)--(u)--(r);
        \else
          \draw[current] (l)--node[amount,above left] {$\topflow$}(u);
          \draw[current] (u)--node[amount,above right] {$\topflow$}(r);
        \fi
        \ifx\bottomflow\zeroflow
          \draw[conductor] (l)--(d)--(r);
        \else
          \draw[current] (l)--node[amount,below left] {$\bottomflow$}(d);
          \draw[current] (d)--node[amount,below right] {$\bottomflow$}(r);
        \fi
        \draw[current] (l)--node[amount,above] {$\midflow$}(r);
        \node at (0,-16) {$\mathcal E=\energy$};
      \end{scope}
    }
  \end{scope}
  \coordinate (base) at ($(panelrow.south)+(0,-3mm)$);
  \foreach \case in {a,b,c}
    \node[panelmark] at (panel\case.center |- base) {(\case)};
\end{tikzpicture}
