% Square vertices 1,3 share a value; circles and squares mark the two classes.
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,
  same/.style={circle,draw=FigureBlue,fill=FigureBlueFill,inner sep=0pt,
    minimum size=4.3mm,line width=.52pt},
  other/.style={rectangle,rounded corners=1.1pt,draw=FigureGreen,fill=FigureGreenFill,
    inner sep=0pt,minimum size=4.3mm,line width=.52pt},
  edge/.style={draw=FigureStroke,line width=.65pt},
  panel/.style={anchor=mid,font=\figurefont\footnotesize}]
  \newcommand{\BlockValueGraph}[2]{%
    \node[same] (one) at (0,14) {$1$};
    \node[other] (two) at (14,14) {$2$};
    \node[same] (three) at (14,0) {$3$};
    \node[other] (four) at (0,0) {$4$};
    \draw[edge] (one) -- (two) -- (three) -- (four) -- (one) -- (three);
    \node[left] at (one.west) {$#1$};
    \node[right] at (two.east) {$#2$};
    \node[right] at (three.east) {$#1$};
    \node[left] at (four.west) {$#2$};
  }
  \begin{scope}
    \BlockValueGraph{a}{b}
    \node[panel] at (7,-7) {(a)};
  \end{scope}
  \draw[edge,->] (24,7) -- node[above] {$A$} (38,7);
  \begin{scope}[xshift=54mm]
    \BlockValueGraph{a+2b}{2a}
    \node[panel] at (7,-7) {(b)};
  \end{scope}
\end{tikzpicture}
