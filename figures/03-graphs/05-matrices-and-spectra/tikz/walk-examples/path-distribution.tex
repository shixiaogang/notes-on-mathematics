% Amounts are unconditional path-flow contributions between time slices.
\begin{tikzpicture}[
  x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
  cell/.style={rounded corners=1.8pt,draw=FigureBlue,fill=FigureBlueFill,
    minimum width=9mm,minimum height=5.2mm,inner sep=1pt,line width=.55pt},
  empty/.style={cell,draw=FigureGrid,fill=white,text=FigureMuted},
  transfer/.style={->,draw=FigureBlue,line width=.65pt},
  amount/.style={fill=white,inner sep=.65pt,text=FigureInk},
  hint/.style={font=\FigureNoteFont\fontsize{7.5}{9}\selectfont,text=FigureMuted}
]
  \foreach \i in {1,...,4} {
    \node[hint] at ({22*\i},8) {顶点 \(\i\)};
  }
  \foreach \t/\masses in {
      0/{1,0,0,0},
      1/{0,1,0,0},
      2/{1/2,0,1/2,0},
      3/{0,3/4,0,1/4}} {
    \node[anchor=west,text=FigureInk] at (0,{-13*\t}) {\(t=\t\)};
    \foreach \mass [count=\i from 1] in \masses {
      \pgfmathparse{\mass}
      \ifdim\pgfmathresult pt=0pt
        \node[empty] (m-\t-\i) at ({22*\i},{-13*\t}) {\(\mass\)};
      \else
        \node[cell] (m-\t-\i) at ({22*\i},{-13*\t}) {\(\mass\)};
      \fi
    }
  }
  \draw[transfer] (m-0-1.south east) -- node[amount] {\(1\)} (m-1-2.north west);
  \draw[transfer] (m-1-2.south west) -- node[amount] {\(1/2\)} (m-2-1.north east);
  \draw[transfer] (m-1-2.south east) -- node[amount] {\(1/2\)} (m-2-3.north west);
  \draw[transfer] (m-2-1.south east) -- node[amount] {\(1/2\)} (m-3-2.north west);
  \draw[transfer] (m-2-3.south west) -- node[amount] {\(1/4\)} (m-3-2.north east);
  \draw[transfer] (m-2-3.south east) -- node[amount] {\(1/4\)} (m-3-4.north west);
\end{tikzpicture}
