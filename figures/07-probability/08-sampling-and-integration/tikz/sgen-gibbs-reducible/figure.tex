% Original analytic teaching diagram, 2026-11-12.
% (a) Uniform target on [0,1]^2 U [2,3]^2. For x_2 in [0,1] the conditional of x_1 is Unif[0,1] (red slice),
%     so axis-parallel Gibbs moves (green path) never leave the lower square.
% (b) Uniform target on [0,2]^2 U [1,3]^2. The squares share the strip 1 <= x_2 <= 2, whose horizontal slice
%     reaches both squares, so a Gibbs path can pass from one square to the other.
\definecolor{VB}{HTML}{7188A3}\definecolor{VBF}{HTML}{E5EAF0}
\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}
\definecolor{VY}{HTML}{B4A16F}\definecolor{VYF}{HTML}{F1ECDD}
\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}
\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=10mm,y=10mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont},
sq/.style={draw=VB,line width=.8pt,fill=VBF},
gp/.style={VG,line width=.8pt},
dot/.style={circle,fill=VG,inner sep=0pt,minimum size=1.4mm}]
% ===== (a) =====
\draw[->,VL,line width=.5pt] (-.2,0)--(3.4,0) node[right,text=VI] {$x_1$};
\draw[->,VL,line width=.5pt] (0,-.2)--(0,3.4) node[above,text=VI] {$x_2$};
\foreach \t in {1,2,3}{\draw[VL,line width=.5pt] (\t,0)--(\t,-.08) node[below,lab] {$\t$}; \draw[VL,line width=.5pt] (0,\t)--(-.08,\t) node[left,lab] {$\t$};}
\fill[sq] (0,0) rectangle (1,1); \fill[sq] (2,2) rectangle (3,3);
\draw[VR,line width=1.1pt] (0,.6)--(1,.6);
\draw[VR,line width=.6pt,dashed] (1,.6)--(3.2,.6);
\draw[gp] (.3,.6)--(.8,.6)--(.8,.2)--(.15,.2)--(.15,.85)--(.55,.85)--(.55,.35);
\foreach \p in {(.3,.6),(.8,.6),(.8,.2),(.15,.2),(.15,.85),(.55,.85),(.55,.35)}{\node[dot] at \p {};}
\node[text=VI] at (1.6,-.75) {(a)};
% ===== (b) =====
\begin{scope}[shift={(4.6,0)}]
\draw[->,VL,line width=.5pt] (-.2,0)--(3.4,0) node[right,text=VI] {$x_1$};
\draw[->,VL,line width=.5pt] (0,-.2)--(0,3.4) node[above,text=VI] {$x_2$};
\foreach \t in {1,2,3}{\draw[VL,line width=.5pt] (\t,0)--(\t,-.08) node[below,lab] {$\t$}; \draw[VL,line width=.5pt] (0,\t)--(-.08,\t) node[left,lab] {$\t$};}
\fill[sq] (0,0) rectangle (2,2); \fill[sq] (1,1) rectangle (3,3);
\draw[VB,line width=.8pt] (0,0) rectangle (2,2); \draw[VB,line width=.8pt] (1,1) rectangle (3,3);
\draw[VR,line width=1.1pt] (0,1.5)--(3,1.5);
\draw[gp] (.4,.5)--(.4,1.5)--(2.6,1.5)--(2.6,2.7)--(1.7,2.7);
\foreach \p in {(.4,.5),(.4,1.5),(2.6,1.5),(2.6,2.7),(1.7,2.7)}{\node[dot] at \p {};}
\node[text=VI] at (1.6,-.75) {(b)};
\end{scope}
\end{tikzpicture}
