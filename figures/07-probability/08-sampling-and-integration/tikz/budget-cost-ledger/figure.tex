% Original analytic teaching diagram, 2026-11-12.
% Cost ledger of a sampling computation: elementary operations make up one iteration; an output sample costs
% M iterations for rejection (k for thinning); an effective sample for a given f costs tau_f outputs for a
% correlated chain.  One-off preparation C_0 and the burn-in b iterations enter the total once.
\definecolor{VB}{HTML}{7188A3}\definecolor{VBF}{HTML}{E5EAF0}
\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}
\definecolor{VY}{HTML}{B4A16F}\definecolor{VYF}{HTML}{F1ECDD}
\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}
\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont},
box/.style={draw=VB,fill=VBF,line width=.6pt,rounded corners=1mm,align=center,text=VI,minimum height=13mm,inner sep=1.2mm},
gbox/.style={draw=VG,fill=VGF,line width=.6pt,rounded corners=1mm,align=center,text=VI,minimum height=9mm,inner sep=1.2mm},
arr/.style={->,draw=VL,line width=.6pt}]
% top row: per-unit costs
\node[box,text width=18mm] (op) at (0,30) {单项操作\\[-.3mm]{\fontsize{7.5}{9}\selectfont 随机数、密度\\[-.6mm] 导数、被积函数}};
\node[box,text width=13mm] (it) at (27,30) {一次迭代\\ $c_{\mathrm{it}}$};
\node[box,text width=17mm] (out) at (58,30) {一个输出样本\\ $c_{\mathrm{out}}$};
\node[box,text width=17mm] (eff) at (88,30) {一个有效样本\\ $c_{\mathrm{eff}}$};
\draw[arr] (op) -- node[above,lab]{求和} (it);
\draw[arr] (it) -- node[above,lab,align=center]{$\times M$\\[-.5mm]或 $\times k$} (out);
\draw[arr] (out) -- node[above,lab]{$\times\tau_{f}$} (eff);
% bottom row: one-off items and the total
\node[gbox,text width=26mm] (prep) at (4,8) {预备计算 $C_{0}$\\[-.3mm]{\fontsize{7.5}{9}\selectfont 分解、试运行、拟合}};
\node[gbox,text width=20mm] (burn) at (38,8) {初始化\\ $b\,c_{\mathrm{it}}$};
\node[gbox,text width=30mm,draw=VR,fill=VRF] (tot) at (80,8) {总成本\\ $C_{0}+b\,c_{\mathrm{it}}+n\,c_{\mathrm{out}}$};
\draw[arr] (prep) -- (burn);
\draw[arr] (burn) -- (tot);
\draw[arr] (out.south) -- ++(0,-5) -| ([xshift=-6mm]tot.north);
\end{tikzpicture}
