% Original analytic teaching diagram, 2026-11-12.
% Two-state space {1,2}, target pi = (1/2, 1/2).  K_A = identity, K_B = all entries 1/2; both leave pi invariant.
% Adaptive rule: use K_A when the current state is 1 and K_B when it is 2.  The combined transition has
% 1 -> 1 w.p. 1, 2 -> 1 w.p. 1/2, 2 -> 2 w.p. 1/2, so state 1 is absorbing.
\definecolor{VB}{HTML}{7188A3}\definecolor{VBF}{HTML}{E5EAF0}
\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}
\definecolor{VY}{HTML}{B4A16F}\definecolor{VYF}{HTML}{F1ECDD}
\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}
\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont},
st/.style={circle,draw=VB,fill=VBF,line width=.6pt,minimum size=7mm,inner sep=0,text=VI},
arr/.style={->,draw=VL,line width=.6pt},
loop/.style={->,draw=VL,line width=.6pt,looseness=6}]
% (a) K_A
\node[st] (a1) at (0,10) {$1$}; \node[st] (a2) at (16,10) {$2$};
\draw[loop] (a1) to[out=120,in=60] node[above,lab]{$1$} (a1);
\draw[loop] (a2) to[out=120,in=60] node[above,lab]{$1$} (a2);
\node[text=VI] at (8,-1) {$K_{A}$};
\node at (8,-7) {(a)};
% (b) K_B
\node[st] (b1) at (34,10) {$1$}; \node[st] (b2) at (50,10) {$2$};
\draw[loop] (b1) to[out=120,in=60] node[above,lab]{$\tfrac12$} (b1);
\draw[loop] (b2) to[out=120,in=60] node[above,lab]{$\tfrac12$} (b2);
\draw[arr] (b1) to[bend left=25] node[above,lab]{$\tfrac12$} (b2);
\draw[arr] (b2) to[bend left=25] node[below,lab]{$\tfrac12$} (b1);
\node[text=VI] at (42,-1) {$K_{B}$};
\node at (42,-7) {(b)};
% (c) adaptive combination
\node[st,draw=VR,fill=VRF] (c1) at (70,10) {$1$}; \node[st] (c2) at (86,10) {$2$};
\draw[loop,draw=VR,line width=.9pt] (c1) to[out=120,in=60] node[above,lab]{$1$} (c1);
\draw[loop] (c2) to[out=120,in=60] node[above,lab]{$\tfrac12$} (c2);
\draw[arr] (c2) to[bend left=25] node[below,lab]{$\tfrac12$} (c1);
\node[text=VI,align=center] at (78,-1) {在 $1$ 用 $K_{A}$，在 $2$ 用 $K_{B}$};
\node at (78,-7) {(c)};
\end{tikzpicture}
