% Original analytic teaching diagram, 2026-11-12.
% Computation graph of one reparameterized sample: eps -> x = T_theta(eps) -> y = f_theta(x).
% theta reaches y along two routes: directly through f (partial derivative of f in theta, yellow),
% and through the generation map (Jacobian of T in theta, then gradient of f in x, green).
% The pathwise derivative is the sum of the two route contributions; eps is held fixed.
\definecolor{VB}{HTML}{7188A3}\definecolor{VBF}{HTML}{E5EAF0}
\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}
\definecolor{VY}{HTML}{B4A16F}\definecolor{VYF}{HTML}{F1ECDD}
\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}
\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont},
var/.style={circle,draw=VL,line width=.5pt,fill=white,minimum size=7mm,inner sep=.5pt,text=VI},
fix/.style={circle,draw=VB,line width=.7pt,fill=VBF,minimum size=7mm,inner sep=.5pt,text=VI},
th/.style={circle,fill=VRF,draw=VR,line width=.6pt,inner sep=.6pt,minimum size=7mm,text=VI},
ar/.style={->,VL,line width=.55pt,shorten >=.4mm,shorten <=.4mm},
ga/.style={->,VG,line width=1pt,shorten >=.4mm,shorten <=.4mm},
ya/.style={->,VY,line width=1.1pt,shorten >=.4mm,shorten <=.4mm}]
\node[fix] (e) at (0,0) {$\varepsilon$};
\node[var] (x) at (34,0) {$x$};
\node[var] (y) at (68,0) {$y$};
\node[th] (t) at (34,-20) {$\bm{\theta}$};
\draw[ar] (e)--(x) node[lab,midway,above=.6mm] {$T_{\bm{\theta}}$};
\draw[ga] (x)--(y) node[lab,midway,above=.6mm,text=VG] {$\nabla_{x}f_{\bm{\theta}}(x)$};
\draw[ga] (t)--(x) node[lab,midway,left=.8mm,text=VG] {$\partial T_{\bm{\theta}}(\varepsilon)/\partial\bm{\theta}$};
\draw[ya] (t)--(y) node[lab,midway,below right=.2mm and .6mm,text=VY] {$\nabla_{\bm{\theta}}f_{\bm{\theta}}(x)$};
\node[lab,anchor=west] at (74,0) {$=f_{\bm{\theta}}(T_{\bm{\theta}}(\varepsilon))$};
\node[lab,anchor=north] at (0,-5) {固定};
\end{tikzpicture}
