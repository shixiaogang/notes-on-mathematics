% Original analytic teaching diagram, 2026-11-12.
% Rows: four outcomes; columns: n = 1..16; a filled cell means omega is in A_n.
% omega_1: n even (infinitely many); omega_2: n in {1,2,4,5} (finitely many, last n = 5);
% omega_3: n a perfect square 1,4,9,16 (infinitely many, sparse); omega_4: never.
% Red line at N = 7: the tail union B_7 = union_{n>=7} A_n contains omega_1 and omega_3 only.
\definecolor{VB}{HTML}{7188A3}\definecolor{VBF}{HTML}{E5EAF0}
\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}
\definecolor{VY}{HTML}{B4A16F}\definecolor{VYF}{HTML}{F1ECDD}
\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}
\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont},
txt/.style={text=VI},
cell/.style={draw=VL,line width=.4pt},
ev/.style={draw=VB,fill=VBF,line width=.6pt}]
\def\w{4.4}\def\h{5}
% grid
\foreach \r in {1,...,4}{\foreach \n in {1,...,16}{\draw[cell] ({(\n-1)*\w},{(4-\r)*\h}) rectangle ++(\w,\h);}}
% omega_1: even n
\foreach \n in {2,4,...,16}{\draw[ev] ({(\n-1)*\w},{3*\h}) rectangle ++(\w,\h);}
% omega_2: 1,2,4,5
\foreach \n in {1,2,4,5}{\draw[ev] ({(\n-1)*\w},{2*\h}) rectangle ++(\w,\h);}
% omega_3: squares
\foreach \n in {1,4,9,16}{\draw[ev] ({(\n-1)*\w},{1*\h}) rectangle ++(\w,\h);}
% row labels
\foreach \r in {1,...,4}{\node[txt,anchor=east] at (-1,{(4-\r)*\h+\h/2}) {$\omega_{\r}$};}
% column labels
\foreach \n in {1,4,7,10,13,16}{\node[lab] at ({(\n-.5)*\w},-2.6) {$\n$};}
\node[txt] at ({8*\w},-7) {$n$};
% tail from N = 7
\draw[VR,line width=1.1pt] ({6*\w},-1) -- ({6*\w},{4*\h+1.4});
\node[text=VR,anchor=south] at ({6*\w},{4*\h+1.2}) {$N=7$};
% right labels
\node[lab,anchor=west] at ({16*\w+1.5},{3*\h+\h/2}) {无穷多次};
\node[lab,anchor=west] at ({16*\w+1.5},{2*\h+\h/2}) {有限次};
\node[lab,anchor=west] at ({16*\w+1.5},{1*\h+\h/2}) {无穷多次};
\node[lab,anchor=west] at ({16*\w+1.5},{\h/2}) {从不发生};
\end{tikzpicture}
