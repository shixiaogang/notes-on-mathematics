% Original analytic teaching diagram, 2026-11-12.
% All 16 sign paths of a Rademacher walk S_0..S_4. Node labels are the numbers of paths through each
% value (binomial coefficients). Threshold a = 3 (red lines). The 4 paths whose running maximum |S_k|
% reaches 3 pass through (3, +-3); only the 2 paths ending at +-4 have |S_4| >= 3.

\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}

\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}
\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont},
nd/.style={circle,draw=VL,fill=white,line width=.5pt,inner sep=0pt,minimum size=4.6mm,text=VI},
hit/.style={circle,draw=VR,fill=VRF,line width=.7pt,inner sep=0pt,minimum size=4.6mm,text=VI}]
% columns k = 0..4 at x = 0, 18, 36, 54, 72; value s at y = 6*s
\fill[VRF] (-6,15) rectangle (80,27);
\fill[VRF] (-6,-27) rectangle (80,-15);
\draw[VR,line width=.8pt] (-6,15) -- (80,15);
\draw[VR,line width=.8pt] (-6,-15) -- (80,-15);
\node[text=VR,anchor=west] at (81,15) {$S=3$};
\node[text=VR,anchor=west] at (81,-15) {$S=-3$};
% edges
\foreach \k/\s in {0/0,1/1,1/-1,2/2,2/0,2/-2,3/1,3/-1}{
  \draw[VL,line width=.5pt] ({18*\k},{6*\s}) -- ({18*(\k+1)},{6*(\s+1)});
  \draw[VL,line width=.5pt] ({18*\k},{6*\s}) -- ({18*(\k+1)},{6*(\s-1)});}
\foreach \s in {3,-3}{
  \draw[VR,line width=.9pt] (54,{6*\s}) -- (72,{6*(\s+1)});
  \draw[VR,line width=.9pt] (54,{6*\s}) -- (72,{6*(\s-1)});}

% nodes with path counts

\node[nd] at (18,6) {$1$};\node[nd] at (18,-6) {$1$};
\node[nd] at (36,12) {$1$};\node[nd] at (36,0) {$2$};\node[nd] at (36,-12) {$1$};
\node[hit] at (54,18) {$1$};\node[nd] at (54,6) {$3$};\node[nd] at (54,-6) {$3$};\node[hit] at (54,-18) {$1$};
\node[hit] at (72,24) {$1$};\node[nd] at (72,12) {$4$};\node[nd] at (72,0) {$6$};\node[nd] at (72,-12) {$4$};\node[hit] at (72,-24) {$1$};
\foreach \k in {0,...,4}{\node[text=VI] at ({18*\k},-31) {$k=\k$};}
\end{tikzpicture}
