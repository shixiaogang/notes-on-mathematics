% Original analytic teaching diagram, 2026-11-12.
% Two fair coins, four equally likely outcomes. Shaded cells in (a), (b), (c):
% A = {first heads}, B = {second heads}, C = {both coins agree}.
% Each event has 2 of 4 cells, each pair shares exactly 1 cell (pairwise independent),
% but A cap B cap C = {HH} has probability 1/4, not 1/8.
\definecolor{VB}{HTML}{7188A3}\definecolor{VBF}{HTML}{E5EAF0}
\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}
\definecolor{VY}{HTML}{B4A16F}\definecolor{VYF}{HTML}{F1ECDD}
\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}
\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=FigureMuted,font=\figurefont\fontsize{7.5}{9}\selectfont},
txt/.style={text=VI},
cell/.style={draw=VL,line width=.5pt},
ev/.style={draw=VB,fill=VBF,line width=.7pt}]
\def\s{9}
% cells: (col,row) col = first coin (1=正,2=反), row = second coin (2=正,1=反)
\foreach \sh/\cells/\name/\tag in {0/{1/1,1/2}/A/a,30/{1/2,2/2}/B/b,60/{1/2,2/1}/C/c,90/{1/2}/{A\cap B\cap C}/d}{
\begin{scope}[shift={(\sh,0)}]
\foreach \i in {1,2}{\foreach \j in {1,2}{\draw[cell] ({(\i-1)*\s},{(\j-1)*\s}) rectangle ++(\s,\s);}}
\foreach \i/\j in \cells {\draw[ev] ({(\i-1)*\s},{(\j-1)*\s}) rectangle ++(\s,\s);}
\node[txt] at ({.5*\s},{1.5*\s}) {正正}; \node[txt] at ({1.5*\s},{1.5*\s}) {反正};
\node[txt] at ({.5*\s},{.5*\s}) {正反}; \node[txt] at ({1.5*\s},{.5*\s}) {反反};
\node[txt] at (\s,-5) {(\tag) $\name$};
\end{scope}}
\end{tikzpicture}
