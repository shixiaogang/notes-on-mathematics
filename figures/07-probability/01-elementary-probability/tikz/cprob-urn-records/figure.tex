% Three sample spaces for drawing two balls without replacement from
% {1,2,3 black; 4,5 white}: ordered pairs (20), unordered pairs (10), colour record (4).
\definecolor{VB}{HTML}{7188A3}\definecolor{VBF}{HTML}{E5EAF0}
\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}
\definecolor{VY}{HTML}{B4A16F}\definecolor{VYF}{HTML}{F1ECDD}
\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}
\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=FigureMuted,font=\figurefont\fontsize{7.5}{9}\selectfont},
txt/.style={text=VI},
cell/.style={draw=VL,line width=.5pt},
ev/.style={draw=VB,fill=VBF,line width=.6pt},
blk/.style={circle,fill=VL,text=white,inner sep=0pt,minimum size=4mm,font=\figurefont\fontsize{7}{8}\selectfont},
wht/.style={circle,draw=VL,line width=.5pt,fill=white,text=VI,inner sep=0pt,minimum size=4mm,font=\figurefont\fontsize{7}{8}\selectfont}]
\def\s{5.5}
% ---------- (a) ordered pairs ----------
\begin{scope}[shift={(0,0)}]
\foreach \i in {1,...,5}{\foreach \j in {1,...,5}{
  \pgfmathsetmacro{\x}{(\j-1)*\s}\pgfmathsetmacro{\y}{(5-\i)*\s}
  \ifnum\i=\j
    \draw[cell,draw=VM] (\x,\y) rectangle ++(\s,\s);
    \draw[VM,line width=.5pt] (\x+1.4,\y+1.4)--(\x+\s-1.4,\y+\s-1.4) (\x+1.4,\y+\s-1.4)--(\x+\s-1.4,\y+1.4);
  \else
    \draw[cell] (\x,\y) rectangle ++(\s,\s);
  \fi}}
\foreach \i/\j in {1/2,1/3,2/1,2/3,3/1,3/2}{\pgfmathsetmacro{\x}{(\j-1)*\s}\pgfmathsetmacro{\y}{(5-\i)*\s}\draw[ev] (\x,\y) rectangle ++(\s,\s);}
\foreach \k in {1,2,3}{\node[blk] at ({(\k-.5)*\s},{5*\s+3.2}) {\k}; \node[blk] at (-3.4,{(5-\k+.5)*\s}) {\k};}
\foreach \k in {4,5}{\node[wht] at ({(\k-.5)*\s},{5*\s+3.2}) {\k}; \node[wht] at (-3.4,{(5-\k+.5)*\s}) {\k};}
\node[lab,anchor=south] at (13.75,{5*\s+5.6}) {第二个球};
\node[lab,rotate=90,anchor=south] at (-6.2,13.75) {第一个球};
\node[txt] at (13.75,-6) {(a)};
\end{scope}
% ---------- (b) unordered pairs ----------
\begin{scope}[shift={(40,0)}]
\foreach \i in {1,...,5}{\foreach \j in {1,...,5}{
  \ifnum\i<\j
    \pgfmathsetmacro{\x}{(\j-1)*\s}\pgfmathsetmacro{\y}{(5-\i)*\s}
    \draw[cell] (\x,\y) rectangle ++(\s,\s);
  \fi}}
\foreach \i/\j in {1/2,1/3,2/3}{\pgfmathsetmacro{\x}{(\j-1)*\s}\pgfmathsetmacro{\y}{(5-\i)*\s}\draw[ev] (\x,\y) rectangle ++(\s,\s);}
\foreach \k in {2,3}{\node[blk] at ({(\k-.5)*\s},{5*\s+3.2}) {\k};}
\foreach \k in {4,5}{\node[wht] at ({(\k-.5)*\s},{5*\s+3.2}) {\k};}
\foreach \k in {1,2,3}{\node[blk] at ({(\k-1)*\s+2.4},{(5-\k+.5)*\s}) {\k};}
\node[wht] at ({3*\s+2.4},{1.5*\s}) {4};
\node[lab,anchor=south] at (19.25,{5*\s+5.6}) {较大编号};
\node[lab,rotate=90,anchor=south] at (-1.5,15) {较小编号};
\node[txt] at (13.75,-6) {(b)};
\end{scope}
% ---------- (c) colour record ----------
\begin{scope}[shift={(79.2,5.5)}]
\def\c{10}
\draw[ev] (0,\c) rectangle ++(\c,\c); \node[txt] at (5,15) {$\tfrac{6}{20}$};
\draw[cell] (\c,\c) rectangle ++(\c,\c); \node[txt] at (15,15) {$\tfrac{6}{20}$};
\draw[cell] (0,0) rectangle ++(\c,\c); \node[txt] at (5,5) {$\tfrac{6}{20}$};
\draw[cell] (\c,0) rectangle ++(\c,\c); \node[txt] at (15,5) {$\tfrac{2}{20}$};

\node[lab] at (-3,15) {黑}; \node[lab] at (-3,5) {白};
\node[txt] at (10,-11.5) {(c)};
\end{scope}
\end{tikzpicture}
