% Two dice as a product sample space. A = {first die in {1,2}} (columns, blue frame),
% B = {second die in {4,5,6}} (rows, yellow frame). A cap B is the 2 x 3 rectangle,
% so |A cap B| = |{1,2}| * |{4,5,6}| = 6 and P(A cap B) = (2/6)(3/6).

\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=FigureMuted,font=\figurefont\fontsize{7.5}{9}\selectfont},

\def\s{6}

\node[txt,anchor=west] at ({6*\s+1},{4.5*\s}) {$B$};
\foreach \k in {1,...,6}{\node[lab] at ({(\k-.5)*\s},-2.8) {\k}; \node[lab] at (-2.8,{(\k-.5)*\s}) {\k};}
