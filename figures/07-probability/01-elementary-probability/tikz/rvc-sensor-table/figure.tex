% Original analytic teaching diagram, 2026-11-12.
% Joint probability table of two sensor levels X (rows) and Y (columns);
% row sums give p_X, column sums give p_Y; diagonal cells form the event {X=Y}.
\definecolor{VB}{HTML}{7188A3}\definecolor{VBF}{HTML}{E5EAF0}
\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}
\definecolor{VY}{HTML}{B4A16F}\definecolor{VYF}{HTML}{F1ECDD}
\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}
\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont}]
\def\cw{13}\def\ch{7}
% diagonal event
\foreach \r in {0,1,2}{
  \fill[VBF] (\r*\cw,-\r*\ch) rectangle (\r*\cw+\cw,-\r*\ch-\ch);
}
% cells
\foreach \r/\a/\b/\c in {0/0.20/0.10/0,1/0.10/0.20/0.10,2/0/0.10/0.20}{
  \node[text=VI] at (0.5*\cw,-\r*\ch-0.5*\ch) {$\a$};
  \node[text=VI] at (1.5*\cw,-\r*\ch-0.5*\ch) {$\b$};
  \node[text=VI] at (2.5*\cw,-\r*\ch-0.5*\ch) {$\c$};
}
\draw[VL!55,line width=.4pt] (0,0) grid[xstep=\cw,ystep=\ch] (3*\cw,-3*\ch);
\draw[VB,line width=.8pt] (0,0) -- (\cw,0) -- (\cw,-\ch) -- (2*\cw,-\ch) -- (2*\cw,-2*\ch) -- (3*\cw,-2*\ch) -- (3*\cw,-3*\ch) -- (2*\cw,-3*\ch) -- (2*\cw,-2*\ch) -- (\cw,-2*\ch) -- (\cw,-\ch) -- (0,-\ch) -- cycle;
% headers
\foreach \k in {0,1,2}{
  \node[text=VI] at (\k*\cw+0.5*\cw,4) {$y=\k$};
  \node[text=VI,left] at (-1,-\k*\ch-0.5*\ch) {$x=\k$};
}
% row sums
\draw[->,VL,line width=.6pt] (3*\cw+1.5,-1.5*\ch) -- (3*\cw+7,-1.5*\ch);
\foreach \r/\s in {0/0.3,1/0.4,2/0.3}{
  \node[text=VI] at (3*\cw+15,-\r*\ch-0.5*\ch) {$\s$};
}
\node[text=VI] at (3*\cw+15,4) {$p_X(x)$};
\draw[VL,line width=.7pt] (3*\cw+9,0) rectangle (3*\cw+21,-3*\ch);
% column sums
\draw[->,VL,line width=.6pt] (1.5*\cw,-3*\ch-1.2) -- (1.5*\cw,-3*\ch-4.5);
\foreach \k/\s in {0/0.3,1/0.4,2/0.3}{
  \node[text=VI] at (\k*\cw+0.5*\cw,-3*\ch-9) {$\s$};
}
\node[text=VI,left] at (-1,-3*\ch-9) {$p_Y(y)$};
\draw[VL,line width=.7pt] (0,-3*\ch-5.5) rectangle (3*\cw,-3*\ch-12.5);
\end{tikzpicture}
