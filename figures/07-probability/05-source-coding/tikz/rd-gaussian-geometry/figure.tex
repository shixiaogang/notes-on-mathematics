% L^2 geometry of the optimal Gaussian test channel, sigma^2=1, D=1/4 (scale 60 mm per unit of
% standard deviation): X = hat X + Z with hat X and Z independent (orthogonal), ||hat X||=sqrt(3)/2,
% ||Z||=1/2, ||X||=1. Projection of hat X on X is aX with a=1-D/sigma^2=3/4.

\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}
\definecolor{VY}{HTML}{B4A16F}\definecolor{VYF}{HTML}{F1ECDD}
\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}
\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont}]
\def\s{60}
\coordinate (O) at (0,0);
\coordinate (H) at ({\s*0.8660254},0);
\coordinate (X) at ({\s*0.8660254},{\s*0.5});

\coordinate (P) at ({0.75*\s*0.8660254},{0.75*\s*0.5});

\draw[->,VG,line width=1.3pt] (O) -- (H) node[midway,below=1.2mm,text=VI] {$\hat X$：$\mathbb{E}\hat X^{2}=\sigma^{2}-D$};
\draw[->,VR,line width=1.3pt] (H) -- (X) node[midway,right=1mm,text=VI] {$Z$：$\mathbb{E}Z^{2}=D$};
\draw[->,VB,line width=1.3pt] (O) -- (X) node[pos=.45,above left=.3mm,text=VI] {$X$：$\mathbb{E}X^{2}=\sigma^{2}$};
\draw[VP,line width=.8pt,densely dashed] (H) -- (P);
\fill[VP] (P) circle (1.0);
\node[text=VP,anchor=south east] at ($(P)+(-.5,.6)$) {$aX$};
