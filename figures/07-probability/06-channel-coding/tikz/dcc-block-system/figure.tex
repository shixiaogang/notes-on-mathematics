% Original analytic teaching diagram, 2026-11-12.
% Block transmission over a discrete memoryless channel: the message m is mapped by the encoder f to a whole block
% x_1..x_n; each symbol passes through the same transition matrix W independently; the decoder g sees y_1..y_n and
% outputs the estimate. The dashed red path from the outputs back to the encoder (feedback) is not used.

\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont},
box/.style={draw=#1,line width=.6pt,rounded corners=1.2mm,minimum height=7mm,inner xsep=1.6mm,text=VI,align=center},
ch/.style={draw=VB,fill=VBF,line width=.6pt,minimum width=9mm,minimum height=4.6mm,inner sep=0pt,text=VI},

\node[text=VI] (m) at (0,0) {$m$};
\node[box=VG,fill=VGF] (f) at (14,0) {编码器\\$f$};
\node[ch] (c1) at (50,9) {$W$};
\node[ch] (c2) at (50,3) {$W$};
\node[text=VM] at (50,-2.6) {$\vdots$};
\node[ch] (cn) at (50,-9) {$W$};
\node[box=VG,fill=VGF] (g) at (86,0) {解码器\\$g$};
\node[text=VI] (mh) at (100,0) {$\hat{m}$};
\draw[arr] (m) -- (f);
\foreach \c/\i in {c1/1,c2/2,cn/n} {

  \draw[arr] (\c.east) -- node[above,text=VI]{$y_{\i}$} ++(14,0) -- ++(0,0) |- (g.west);
