% Original analytic teaching diagram, 2026-11-12.
% Transition diagrams of three binary-input channels: (a) binary symmetric channel with flip probability p,
% (b) binary erasure channel with erasure probability epsilon (output e visible to the receiver),
% (c) Z-channel in which input 0 is received correctly and input 1 becomes 0 with probability 1/2.
% Each arrow carries the conditional probability W(y|x); arrows of each input sum to one.
\definecolor{VB}{HTML}{7188A3}\definecolor{VBF}{HTML}{E5EAF0}
\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}
\definecolor{VY}{HTML}{B4A16F}\definecolor{VYF}{HTML}{F1ECDD}
\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}
\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont},
sym/.style={circle,draw=VL,line width=.6pt,minimum size=5.2mm,inner sep=0pt,text=VI,fill=white},
ok/.style={->,draw=VB,line width=.8pt,shorten >=.4mm,shorten <=.4mm},
bad/.style={->,draw=VR,line width=.8pt,shorten >=.4mm,shorten <=.4mm},
era/.style={->,draw=VY,line width=.9pt,shorten >=.4mm,shorten <=.4mm}]
% ---------- (a) BSC
\begin{scope}
\node[sym] (a0) at (0,14) {$0$}; \node[sym] (a1) at (0,0) {$1$};
\node[sym] (b0) at (24,14) {$0$}; \node[sym] (b1) at (24,0) {$1$};
\draw[ok] (a0) -- node[above,text=VB]{$1-p$} (b0);
\draw[ok] (a1) -- node[below,text=VB]{$1-p$} (b1);
\draw[bad] (a0) -- (b1); \draw[bad] (a1) -- (b0);
\node[text=VR] at (6.5,12.6) {$p$}; \node[text=VR] at (6.5,1.4) {$p$};
\node[lab] at (0,21) {$X$}; \node[lab] at (24,21) {$Y$};
\node at (12,-10) {(a)};
\end{scope}
% ---------- (b) BEC
\begin{scope}[xshift=40mm]
\node[sym] (a0) at (0,14) {$0$}; \node[sym] (a1) at (0,0) {$1$};
\node[sym] (b0) at (24,14) {$0$}; \node[sym,draw=VY] (be) at (24,7) {$\mathrm{e}$}; \node[sym] (b1) at (24,0) {$1$};
\draw[ok] (a0) -- node[above,text=VB]{$1-\varepsilon$} (b0);
\draw[ok] (a1) -- node[below,text=VB]{$1-\varepsilon$} (b1);
\draw[era] (a0) -- (be); \draw[era] (a1) -- (be);
\node[text=VY] at (12.5,12.2) {$\varepsilon$}; \node[text=VY] at (12.5,1.8) {$\varepsilon$};
\node[lab] at (0,21) {$X$}; \node[lab] at (24,21) {$Y$};
\node at (12,-10) {(b)};
\end{scope}
% ---------- (c) Z-channel
\begin{scope}[xshift=80mm]
\node[sym] (a0) at (0,14) {$0$}; \node[sym] (a1) at (0,0) {$1$};
\node[sym] (b0) at (24,14) {$0$}; \node[sym] (b1) at (24,0) {$1$};
\draw[ok] (a0) -- node[above,text=VB]{$1$} (b0);
\draw[ok] (a1) -- node[below,text=VB]{$\tfrac12$} (b1);
\draw[bad] (a1) -- (b0);
\node[text=VR] at (8,8.6) {$\tfrac12$};
\node[lab] at (0,21) {$X$}; \node[lab] at (24,21) {$Y$};
\node at (12,-10) {(c)};
\end{scope}
\end{tikzpicture}
