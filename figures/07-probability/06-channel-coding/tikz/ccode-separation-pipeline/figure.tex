% Original analytic teaching diagram, 2026-11-12.
% Separated source-channel transmission: k source symbols -> source encoder -> index m -> channel encoder
% -> n channel inputs -> memoryless channel -> n outputs -> channel decoder -> index estimate -> source decoder
% -> reconstruction of the k symbols. The two failure events are marked in red: the source code fails
% (block outside the indexed set / distortion too large) and the channel decoder returns a wrong index.

\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont},

% top row: transmitter
\node[sb] (se) at (14,20) {信源编码};

\node[cb] (cd) at (86,0) {信道解码};
\node[sb] (sd) at (50,0) {信源解码};
