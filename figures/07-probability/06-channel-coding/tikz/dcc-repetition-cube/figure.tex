% Original analytic teaching diagram, 2026-11-12.
% The eight received blocks of length 3 as vertices of a cube; edges join blocks differing in one position.
% Codewords 000 and 111 sit at opposite corners. Majority decoding assigns the blue vertices (at most one 1)
% to message 0 and the yellow vertices (at least two 1s) to message 1. The red arrow shows the worked example:
% 111 is sent, the second symbol is flipped, 101 is received and decoded as 1.
\definecolor{VB}{HTML}{7188A3}\definecolor{VBF}{HTML}{E5EAF0}
\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}
\definecolor{VY}{HTML}{B4A16F}\definecolor{VYF}{HTML}{F1ECDD}
\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}
\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont},
v0/.style={draw=VB,fill=VBF,line width=.6pt,rounded corners=1mm,minimum width=8.5mm,minimum height=5mm,inner sep=0pt,text=VI},
v1/.style={draw=VY,fill=VYF,line width=.6pt,rounded corners=1mm,minimum width=8.5mm,minimum height=5mm,inner sep=0pt,text=VI},
cw/.style={line width=1.3pt},
ed/.style={draw=VL,line width=.5pt}]
% cube coordinates: bit1 -> x, bit2 -> depth (up-right), bit3 -> y
\node[v0,cw] (000) at (0,0) {$000$};
\node[v0] (100) at (34,0) {$100$};
\node[v0] (010) at (14,10) {$010$};
\node[v1] (110) at (48,10) {$110$};
\node[v0] (001) at (0,26) {$001$};
\node[v1] (101) at (34,26) {$101$};
\node[v1] (011) at (14,36) {$011$};
\node[v1,cw] (111) at (48,36) {$111$};
\foreach \a/\b in {000/100,000/010,000/001,100/110,100/101,010/110,010/011,001/101,001/011,110/111,101/111,011/111}
  \draw[ed] (\a) -- (\b);
\draw[->,draw=VR,line width=1.1pt,shorten >=.6mm,shorten <=.6mm] (111) to[bend left=18] (101);
\node[text=VR,anchor=west] at (44,28.4) {翻转第2位};
\node[lab,anchor=north] at (0,-3.5) {码字};
\node[lab,anchor=south] at (48,39) {码字};
% legend
\node[v0,minimum width=5mm,minimum height=3.4mm] at (66,8) {};
\node[lab,anchor=west] at (69.5,8) {判为 $0$};
\node[v1,minimum width=5mm,minimum height=3.4mm] at (66,2) {};
\node[lab,anchor=west] at (69.5,2) {判为 $1$};
\end{tikzpicture}
