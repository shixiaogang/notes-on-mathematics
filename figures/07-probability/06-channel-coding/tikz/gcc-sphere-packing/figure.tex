% Original analytic teaching diagram, 2026-11-12.
% Sphere-packing heuristic drawn in the plane (n = 2) with P/sigma^2 = 8, unit sigma = 6 mm:
% codewords (blue dots) lie in the power ball of radius sqrt(nP) (blue circle, 16.97 mm), each output lies near a
% noise ball of radius sqrt(n sigma^2) (yellow, 6 mm) around its codeword, and all outputs lie in the ball of radius
% sqrt(n(P + sigma^2)) (purple dashed, 18 mm). Seven non-overlapping noise balls fit; the volume ratio is (18/6)^2 = 9.
\definecolor{VB}{HTML}{7188A3}\definecolor{VBF}{HTML}{E5EAF0}
\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}
\definecolor{VY}{HTML}{B4A16F}\definecolor{VYF}{HTML}{F1ECDD}
\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}
\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont}]
\draw[VP,line width=.8pt,dash pattern=on 2.2mm off 1.2mm] (0,0) circle (18);
\foreach \a in {0,60,...,300}{
  \fill[VYF] (\a:12) circle (6);
  \draw[VY,line width=.6pt] (\a:12) circle (6);
}
\fill[VYF] (0,0) circle (6);
\draw[VY,line width=.6pt] (0,0) circle (6);
\draw[VB,line width=.9pt] (0,0) circle (16.97);
\foreach \a in {0,60,...,300}{\fill[VB] (\a:12) circle (.75);}
\fill[VB] (0,0) circle (.75);
% radii and labels
\draw[->,VL,line width=.5pt] (300:12) -- ++(6,0);
\node[text=VI,anchor=west] at (21,-16) {$\sqrt{n\sigma^2}$};
\draw[VL,line width=.4pt] (20.6,-15.6) -- ($(300:12)+(4.5,-.3)$);
\node[text=VB,anchor=east] at (-21,-13) {$\sqrt{nP}$};
\draw[VL,line width=.4pt] (-20.6,-12.6) -- (-145:16.97);
\node[text=VP,anchor=east] at (-21,13) {$\sqrt{n(P+\sigma^2)}$};
\draw[VL,line width=.4pt] (-20.6,12.6) -- (145:18);
\end{tikzpicture}
