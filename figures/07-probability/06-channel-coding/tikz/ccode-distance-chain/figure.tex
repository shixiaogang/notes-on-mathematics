% Original analytic teaching diagram, 2026-11-12.
% (a) Repetition code of length 5: the codewords 00000 and 11111 are joined by a chain of five single flips.
% Words within distance 2 of a codeword (blue / yellow) are decoded to it; the two radius-2 balls do not meet.
% (b) Repetition code of length 4: the middle word 1100 is at distance 2 from both codewords (red), so 2 flips
% can be detected but not corrected.

\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}
\definecolor{VY}{HTML}{B4A16F}\definecolor{VYF}{HTML}{F1ECDD}
\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}
\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont},
w0/.style={draw=VB,fill=VBF,line width=.6pt,rounded corners=1mm,minimum width=10mm,minimum height=5mm,inner sep=0pt,text=VI},
w1/.style={draw=VY,fill=VYF,line width=.6pt,rounded corners=1mm,minimum width=10mm,minimum height=5mm,inner sep=0pt,text=VI},
wm/.style={draw=VR,fill=VRF,line width=.6pt,rounded corners=1mm,minimum width=10mm,minimum height=5mm,inner sep=0pt,text=VI},
cw/.style={line width=1.3pt},
ed/.style={draw=VL,line width=.5pt}]
% (a)
\foreach \w/\x/\s in {00000/0/w0,10000/16/w0,11000/32/w0,11100/48/w1,11110/64/w1,11111/80/w1}
  \node[\s] (a\w) at (\x,20) {$\w$};
\node[w0,cw] at (0,20) {$00000$};
\node[w1,cw] at (80,20) {$11111$};
\foreach \a/\b in {00000/10000,10000/11000,11000/11100,11100/11110,11110/11111}
  \draw[ed] (a\a) -- (a\b);
\draw[VB,line width=.6pt] (-5,26) -- (-5,27.5) -- (37,27.5) -- (37,26);
\node[text=VB] at (16,30.5) {距离 $\leq2$，判为 $00000$};
\draw[VY,line width=.6pt] (43,26) -- (43,27.5) -- (85,27.5) -- (85,26);
\node[text=VY] at (64,30.5) {距离 $\leq2$，判为 $11111$};
\foreach \k in {1,...,5} \node[lab] at ({16*\k-8},16.3) {$\k$};
\node[lab,anchor=west] at (88,20) {$d=5$};
% (b)
\foreach \w/\x/\s in {0000/8/w0,1000/24/w0,1100/40/wm,1110/56/w1,1111/72/w1}
  \node[\s] (b\w) at (\x,4) {$\w$};
\node[w0,cw] at (8,4) {$0000$};
\node[w1,cw] at (72,4) {$1111$};
\foreach \a/\b in {0000/1000,1000/1100,1100/1110,1110/1111}
  \draw[ed] (b\a) -- (b\b);
\node[text=VR,anchor=north] at (40,0.6) {两边距离都是 $2$};
\node[lab,anchor=west] at (88,4) {$d=4$};
