% Original analytic teaching diagram, 2026-11-12.
% Schematic of decoding spheres in Hamming space. Codewords (blue dots) have pairwise distance at least d_min;
% balls of radius t = floor((d_min - 1)/2) around them (light blue) are disjoint. A received word y1 inside a ball
% is decoded to its centre; a received word y2 outside every ball is recognised as an error (detected) but no
% correction is guaranteed. Up to d_min - 1 flips from a codeword cannot reach another codeword (dashed arc).
\definecolor{VB}{HTML}{7188A3}\definecolor{VBF}{HTML}{E5EAF0}
\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}
\definecolor{VY}{HTML}{B4A16F}\definecolor{VYF}{HTML}{F1ECDD}
\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}
\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont},
ball/.style={draw=VB,fill=VBF,line width=.6pt},
cw/.style={circle,fill=VB,inner sep=0pt,minimum size=2.2mm}]
\foreach \n/\x/\y in {A/0/0,B/30/4,C/14/24,D/46/27,E/62/2}
  {\draw[ball] (\x,\y) circle (11);
   \node[cw] (\n) at (\x,\y) {};}
% radius t
\draw[VL,line width=.5pt,->] (A) -- ++(-7.8,-7.8);
\node[lab] at (-8.5,-3) {$t$};
% distance between A and B
\draw[VL,line width=.5pt,<->] (A) -- (B);
\node[lab,anchor=south] at (15,2.3) {$\geq d_{\min}$};
% received word inside a ball
\node[circle,fill=VR,inner sep=0pt,minimum size=2mm] (y1) at (41,19) {};
\draw[VR,line width=.8pt,->,shorten >=1.2mm] (y1) -- (D);
\node[text=VR,anchor=east] at (39.6,19) {$y_1$};
% received word between balls
\node[circle,draw=VR,fill=white,line width=.8pt,inner sep=0pt,minimum size=2mm] (y2) at (45,7) {};
\node[text=VR,anchor=west] at (46.2,7) {$y_2$};
\end{tikzpicture}
