% Original analytic teaching diagram, 2026-11-12.
% Dependence structure of block transmission without feedback: the message M determines the whole input block
% (the inputs may be dependent through M), while each output Y_i is produced from X_i alone by the same W.
% The blue band groups the outputs: they can be dependent through the inputs, so only H(Y^n) <= sum H(Y_i) holds;
% the conditional entropies add exactly because of memorylessness.
\definecolor{VB}{HTML}{7188A3}\definecolor{VBF}{HTML}{E5EAF0}
\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}
\definecolor{VY}{HTML}{B4A16F}\definecolor{VYF}{HTML}{F1ECDD}
\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}
\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont},
nd/.style={circle,draw=VL,line width=.6pt,minimum size=6.4mm,inner sep=0pt,text=VI,fill=white},
arr/.style={->,draw=VL,line width=.6pt,shorten >=.3mm},
wa/.style={->,draw=VB,line width=.8pt,shorten >=.3mm}]
\fill[VBF,rounded corners=2mm] (42,-21) rectangle (52,21);
\node[nd,draw=VG,fill=VGF] (M) at (0,0) {$M$};
\node[nd] (x1) at (22,15) {$X_1$}; \node[nd] (x2) at (22,5) {$X_2$}; \node[text=VM] at (22,-3.5) {$\vdots$}; \node[nd] (xn) at (22,-15) {$X_n$};
\node[nd] (y1) at (47,15) {$Y_1$}; \node[nd] (y2) at (47,5) {$Y_2$}; \node[text=VM] at (47,-3.5) {$\vdots$}; \node[nd] (yn) at (47,-15) {$Y_n$};
\foreach \x in {x1,x2,xn} \draw[arr] (M) -- (\x);
\draw[wa] (x1) -- node[above,text=VB]{$W$} (y1);
\draw[wa] (x2) -- node[above,text=VB]{$W$} (y2);
\draw[wa] (xn) -- node[above,text=VB]{$W$} (yn);
\node[lab,anchor=west] at (55,12) {$H(Y^n\mid X^n)=\sum_i H(Y_i\mid X_i)$};
\node[lab,anchor=west] at (55,-12) {$H(Y^n)\leq\sum_i H(Y_i)$};
\end{tikzpicture}
