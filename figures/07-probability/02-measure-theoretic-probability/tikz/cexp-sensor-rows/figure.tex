% Original analytic teaching diagram, 2026-11-12.
% Sensor table as a sample space of nine cells. (a) cell probabilities, rows outlined as the
% groups of sigma(X); zero-probability cells in light red. (b) the random variable E[Y|X]:
% every cell of row x carries the row mean 1/3, 1, 5/3 of Y.
\definecolor{VB}{HTML}{7188A3}\definecolor{VBF}{HTML}{E5EAF0}
\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}
\definecolor{VY}{HTML}{B4A16F}\definecolor{VYF}{HTML}{F1ECDD}
\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}
\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont}]
\def\cw{11}\def\ch{7}
% ---------- (a) probabilities ----------
\begin{scope}
\fill[VRF] (2*\cw,0) rectangle (3*\cw,-\ch);
\fill[VRF] (0,-2*\ch) rectangle (\cw,-3*\ch);
\foreach \r/\a/\b/\c in {0/0.2/0.1/0,1/0.1/0.2/0.1,2/0/0.1/0.2}{
  \node[text=VI] at (0.5*\cw,-\r*\ch-0.5*\ch) {$\a$};
  \node[text=VI] at (1.5*\cw,-\r*\ch-0.5*\ch) {$\b$};
  \node[text=VI] at (2.5*\cw,-\r*\ch-0.5*\ch) {$\c$};
}
\draw[VL!45,line width=.4pt] (0,0) grid[xstep=\cw,ystep=\ch] (3*\cw,-3*\ch);
\foreach \r in {0,1,2}{\draw[VB,line width=.9pt] (0.4,-\r*\ch-0.4) rectangle (3*\cw-0.4,-\r*\ch-\ch+0.4);}
\foreach \k in {0,1,2}{
  \node[text=VI] at (\k*\cw+0.5*\cw,3.5) {$y=\k$};
  \node[text=VI,left] at (-1,-\k*\ch-0.5*\ch) {$x=\k$};
}
\node[lab] at (1.5*\cw,-3*\ch-4) {$\mathbb{P}(X=x,\,Y=y)$};
\node[text=VI] at (1.5*\cw,-3*\ch-10) {(a)};
\end{scope}
% ---------- (b) E[Y|X] ----------
\begin{scope}[xshift=52mm]
\foreach \r/\v in {0/{1/3},1/{1},2/{5/3}}{
  \foreach \k in {0,1,2}{\node[text=VG] at (\k*\cw+0.5*\cw,-\r*\ch-0.5*\ch) {$\v$};}
}
\draw[VL!45,line width=.4pt] (0,0) grid[xstep=\cw,ystep=\ch] (3*\cw,-3*\ch);
\foreach \r in {0,1,2}{\draw[VB,line width=.9pt] (0.4,-\r*\ch-0.4) rectangle (3*\cw-0.4,-\r*\ch-\ch+0.4);}
\foreach \k in {0,1,2}{
  \node[text=VI] at (\k*\cw+0.5*\cw,3.5) {$y=\k$};
}
\foreach \r/\p in {0/0.3,1/0.4,2/0.3}{\node[lab,right] at (3*\cw+1,-\r*\ch-0.5*\ch) {$\p$};}
\node[lab] at (3*\cw+5,3.5) {$p_X$};
\node[lab] at (1.5*\cw,-3*\ch-4) {$\mathbb{E}[Y\mid X]$};
\node[text=VI] at (1.5*\cw,-3*\ch-10) {(b)};
\end{scope}
\end{tikzpicture}
