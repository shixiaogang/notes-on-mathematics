% Original analytic teaching diagram, 2026-11-12.
% Conditional expectation as orthogonal projection in L^2 (schematic).
% The plane is the closed subspace L^2(G) of G-measurable square-integrable variables; the line inside it
% is L^2(H) for a smaller H. X projects to E[X|G] (residual perpendicular to the plane); another
% candidate W in the plane gives a longer error (red), with a right angle at E[X|G].
% Projecting E[X|G] further onto the line gives E[X|H] (tower property).
\definecolor{VB}{HTML}{7188A3}\definecolor{VBF}{HTML}{E5EAF0}
\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}
\definecolor{VY}{HTML}{B4A16F}\definecolor{VYF}{HTML}{F1ECDD}
\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}
\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont}]
% plane L^2(G)
\fill[VBF] (0,0) -- (62,0) -- (80,20) -- (18,20) -- cycle;
\draw[VB,line width=.6pt] (0,0) -- (62,0) -- (80,20) -- (18,20) -- cycle;
\node[lab] at (70,3) {$L^{2}(\mathcal{G})$};
% line L^2(H)
\draw[VP,line width=.9pt] (4,3) -- (74,13);
\node[lab,text=VP] at (8,7.5) {$L^{2}(\mathcal{H})$};
% points
\coordinate (O) at (14,4.43);
\coordinate (Z) at (44,12);
\coordinate (X) at (44,40);
\coordinate (W) at (60,8);
% E[X|H]: projection of Z onto the line through (4,3) with direction (70,10)
\coordinate (ZH) at (44.46,8.78);
\fill[VB] (X) circle (1.0);
\draw[VL,line width=.6pt,dashed] (X) -- (Z);
\draw[VR,line width=.9pt] (X) -- (W);
\draw[VL,line width=.5pt] (Z) -- (W);
\draw[VL,line width=.5pt,dashed] (Z) -- (ZH);
% right angle at Z (vertical residual vs in-plane segment)

\node[text=VI,right] at (61,7) {$W$};
\node[text=VI,below left] at (O) {$0$};
\node[text=VR,right] at (53,26) {$X-W$};
\node[text=VI,left] at (43.5,28) {$X-\mathbb{E}[X\mid\mathcal{G}]$};
\end{tikzpicture}
