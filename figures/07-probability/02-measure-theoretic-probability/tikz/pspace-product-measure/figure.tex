% Original analytic teaching diagram, 2026-11-12.
% Product of X ~ Unif(0,1) (horizontal) and Y ~ Exp(1) (vertical, drawn up to y = 3).
% (a) rectangle A x B with A=[0,1/2], B=(1,inf): probability (1/2) e^{-1}.
% (b) region {y > x}; the vertical slice at x has Y-mass e^{-x}, integrating over x in [0,1] gives 1 - e^{-1}.

\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}

\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}

\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=VM,font=\figurefont\fontsize{7.5}{9}\selectfont}]
\def\sx{36}\def\sy{10}
% ---------- (a) ----------
\begin{scope}
\fill[VBF] (0,{\sy}) rectangle ({\sx/2},{3*\sy});
\draw[VB,line width=.8pt] (0,{3*\sy}) -- (0,{\sy}) -- ({\sx/2},{\sy}) -- ({\sx/2},{3*\sy});
\draw[VL,line width=.6pt] (0,0) rectangle (\sx,{3*\sy});

\foreach \x/\t in {0/0,0.5/{1/2},1/1} \node[below,text=VI] at ({\sx*\x},-2) {$\t$};
\foreach \y in {0,1,2,3} \node[left,text=VI] at (-2,{\y*\sy}) {$\y$};
\node[text=VI] at ({\sx/4},{2*\sy}) {$A\times B$};
\node[text=VI] at ({\sx/2},-9) {$x$};
\node[text=VI] at (-9,{1.5*\sy}) {$y$};
\node[text=VI] at ({\sx/2},-15) {(a)};
\end{scope}
% ---------- (b) ----------
\begin{scope}[shift={(54,0)}]

\node[anchor=west,text=VI] at ({0.6*\sx+1},{2.2*\sy}) {$\mathrm{e}^{-x}$};
\foreach \x/\t in {0/0,0.6/x,1/1} \node[below,text=VI] at ({\sx*\x},-2) {$\t$};
\foreach \y in {0,1,2,3} \node[left,text=VI] at (-2,{\y*\sy}) {$\y$};
\node[text=VI] at ({0.78*\sx},{0.35*\sy}) {$y=x$};
\node[text=VI] at ({\sx/2},-9) {$x$};
\node[text=VI] at (-9,{1.5*\sy}) {$y$};
\node[text=VI] at ({\sx/2},-15) {(b)};
\end{scope}
\end{tikzpicture}
