% Original analytic teaching diagram, 2026-11-12.
% Candidate construction for N = 4: J uniform on {1,...,4} (here J = 3, red);
% Y_J is drawn from the conditional law of Y given X (blue), the other candidates from the
% marginal P_Y (grey), independently given (X,J). A score s(X,Y_j) for each candidate is
% normalised by softmax into a guess q(j) of the hidden position.
\definecolor{VB}{HTML}{7188A3}\definecolor{VBF}{HTML}{E5EAF0}
\definecolor{VG}{HTML}{829C8B}\definecolor{VGF}{HTML}{E8EEE9}
\definecolor{VY}{HTML}{B4A16F}\definecolor{VYF}{HTML}{F1ECDD}
\definecolor{VR}{HTML}{A7787D}\definecolor{VRF}{HTML}{F0E5E6}
\definecolor{VP}{HTML}{9588A5}\definecolor{VPF}{HTML}{ECE8F0}
\definecolor{VI}{HTML}{3B4252}\definecolor{VM}{HTML}{6B7280}\definecolor{VL}{HTML}{4C4D4F}
\begin{tikzpicture}[x=1mm,y=1mm,>=Stealth,font=\figurefont\fontsize{8}{10}\selectfont,
lab/.style={text=FigureMuted,font=\figurefont\fontsize{7.5}{9}\selectfont},
txt/.style={text=VI},
bx/.style={draw=VL,line width=.6pt,rounded corners=1.5pt,minimum height=6mm,minimum width=11mm,inner sep=1pt,text=VI},
hid/.style={draw=VR,fill=VRF,line width=.9pt,rounded corners=1.5pt,minimum height=6mm,minimum width=11mm,inner sep=1pt,text=VI},
src/.style={draw=VB,fill=VBF,line width=.7pt,rounded corners=1.5pt,minimum height=6mm,inner sep=2pt,text=VI},
ar/.style={->,draw=VL,line width=.6pt},
arb/.style={->,draw=VB,line width=.8pt}]
\node[src] (x) at (0,0) {$X$};
\node[src] (cond) at (0,-14) {$P_{Y\mid X}$};
\node[bx,draw=VM,text=VI] (marg) at (0,-30) {$P_Y$};
\node[bx] (y1) at (24,6) {$Y_1$};
\node[bx] (y2) at (24,-6) {$Y_2$};
\node[hid] (y3) at (24,-18) {$Y_3$};
\node[bx] (y4) at (24,-30) {$Y_4$};
\draw[arb] (x)--(cond);
\draw[arb] (cond.east)--(y3.west);
\draw[ar,draw=VM] (marg.east)--(y1.west);
\draw[ar,draw=VM] (marg.east)--(y2.west);
\draw[ar,draw=VM] (marg.east)--(y4.west);
\node[lab,text=VR] at (24,-38) {隐藏位置$J=3$};
\foreach \j/\y in {1/6,2/-6,3/-18,4/-30}{
  \node[txt] (s\j) at (46,\y) {$s(X,Y_{\j})$};
  \draw[ar] (y\j.east)--(s\j.west);}
\node[src,minimum height=46mm,minimum width=12mm,align=center] (sm) at (70,-12) {归\\一\\化};
\foreach \j in {1,2,3,4}{\draw[ar] (s\j.east)--(sm.west|-s\j);}
\node[txt] (q) at (86,-12) {$q(j)$};
\draw[ar] (sm.east)--(q.west);
\end{tikzpicture}
